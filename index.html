# Wraps portfolio.html (artifact body) into a standalone index.html for GitHub Pages. from pathlib import Path body = Path("portfolio.html").read_text(encoding="utf-8") HEAD = """ """ TAIL = """ """ # split: everything up to the first HTML comment marker is head-safe (title/link/style) marker = "" head_part, body_part = body.split(marker, 1) body_part = marker + body_part out = HEAD + head_part.strip() + "\n\n\n" + body_part.strip() + TAIL Path("index.html").write_text(out, encoding="utf-8") print("index.html written:", len(out), "bytes")
